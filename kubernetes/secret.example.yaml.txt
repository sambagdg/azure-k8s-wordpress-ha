# Exemple de structure — le vrai Secret est créé par :
# kubectl create secret generic mysql-secret -n gandega-project1 --from-env-file=.secrets/mysql.env
apiVersion: v1
kind: Secret
metadata:
  name: mysql-secret
  namespace: gandega-project1
type: Opaque
stringData:
  MYSQL_ROOT_PASSWORD: "<généré>"
  MYSQL_PASSWORD: "<généré>"
